\section{Single Pendulum}
Our single pendulum model will consist of a point mass of mass $m$ connected to a pivot by a massless rod of length $l$. The angle of the rod with respect to the vertical is defined as $\theta$.
\begin{figure}[h]
    \centering
    \begin{tikzpicture}[thick, >=latex]
        \def\Lone{3}
        \def\thetaone{30}
        \def\r{0.2}

        \coordinate (O) at (0,0);
        \coordinate (M) at ({\Lone * sin(\thetaone)}, {-\Lone * cos(\thetaone)});

        \fill [pattern = north east lines] (-1, 0) rectangle (1,0.2);
        \draw (-1, 0) -- (1, 0);

        \draw[dashed] (O) -- ++(0,-2);

        \draw (O) -- (M) node[midway, right= 0.2cm] {$l$};

        \filldraw[fill=blue!30] (M) circle (\r) node[right=0.3cm] {$m$};

        \draw[->] (0,-1.5) arc (-90:{-90+\thetaone}:1.5) node[midway, below] {$\theta$};
    \end{tikzpicture}
    \caption{Single Pendulum}
    \label{fig:Single Pendulum}
\end{figure}

The vector from the origin to the point mass is defined as $\mathbf{r}$, where it and its first and second derivatives are

\begin{align*}
    \mathbf{r} &= l \sin(\theta) \hat{\imath} - l \cos(\theta) \hat{\jmath} \\
    \dot{\mathbf{r}} &= \dot{\theta} l \cos(\theta) \hat{\imath} + \dot{\theta} l \sin(\theta) \hat{\jmath} \\
    \ddot{\mathbf{r}} &= \left( \ddot{\theta} l \cos(\theta) - \dot{\theta}^2 l \sin(\theta) \right) \hat{\imath} + \left( \ddot{\theta} l \sin(\theta) + \dot{\theta}^2 l \cos(\theta) \right) \hat{\jmath}
\end{align*}

\begin{figure}[h]
    \centering
    \begin{tikzpicture} [thick, >=latex]
        \def\Lone{3}
        \def\thetaone{30}
        \def\r{0.2}

        \coordinate (O) at (0,0);
        \coordinate (M) at ({\Lone * sin(\thetaone)}, {-\Lone * cos(\thetaone)});

        \draw (O) -- (M) node[midway, right= 0.2cm] {$l$};

        \fill (O) circle (0.05);
        \draw[thick] (O) circle (0.1) node[above left] {$O$};

        \filldraw[fill=blue!30] (M) circle (\r) node[above=0.3cm, right=0.2cm] {$m$};

        \draw[->, red, very thick] ($(O)+(0.2,0)$) -- ++(1,0) node[above] {$R_x$};
        \draw[->, red, very thick] ($(O)+(0,0.2)$) -- ++(0,1) node[above right] {$R_y$};
        
        \draw[->, red, very thick] ($(M)+(0.1,-0.3)$) -- ++(0,-1) node[below right] {$mg$};
        \draw[->, red, very thick] ($(M)+(180:0.3)$) -- ++(180:1) node[left] {$m \ddot{r}_x$};
        \draw[->, red, very thick] ($(M)+(-0.1,-0.3)$) -- ++(-90:1) node[below left] {$m \ddot{r}_y$};
    \end{tikzpicture}
    \caption{Free Body Diagram of Single Pendulum}
    \label{fig:Pendulum FBD}
\end{figure}

Figure~\ref{fig:Pendulum FBD} shows a free body diagram of the mass and rod. Using D'Alembert's principle, inertial forces are added inverse to acceleration (components of $\mathbf{r}$) due to rotation. Reaction forces at the pivot are also added

Because reaction forces are unknown, the summation of moments about the pivot is calculated. Substitution values for the components of $\mathbf{r}$ gives us

\[
    \sum M^*_O = 0 = - ml \cos(\theta) \left(  \ddot{\theta} l \cos(\theta) - \dot{\theta}^2 l \sin(\theta) \right) - ml \sin(\theta) \left( \ddot{\theta} l \sin(\theta) + \dot{\theta}^2 l \cos(\theta) \right) - l \sin(\theta) m g 
\]
or simplifying and solving for $\ddot{\theta}$:

\[
    0 = \ddot{\theta} l \left( \cos^2(\theta) + \sin^2(\theta)\right) + \sin(\theta)g
\]

\[
    \ddot{\theta} = -\frac{g}{l} \sin(\theta)
\]

\section{Double Pendulum}
The double pendulum consists of two masses $m_1$ and $m_2$ attached by rigid massless rods of lengths $l_1$ and $l_2$.

\begin{figure}[h]
    \centering
    \begin{tikzpicture}[thick, >=latex]

        \def\Lone{3}
        \def\Ltwo{3}
        \def\thetaone{60} % degrees
        \def\thetatwo{20} % degrees
        \def\r{0.2} % ball radius

        % Coordinates
        \coordinate (O) at (0,0);
        \coordinate (M1) at ({\Lone*sin(\thetaone)}, {-\Lone*cos(\thetaone)});
        \coordinate (M2) at ($(M1) + ({\Ltwo*sin(\thetatwo)}, {-\Ltwo*cos(\thetatwo)})$);

        \draw[dashed] (M1) -- ++(0,-2);

        % Rods
        \draw[thick] (O) -- (M1) node[midway, above right] {$l_1$};
        \draw[thick] (M1) -- (M2) node[midway, right] {$l_2$};

        % Masses
        \filldraw[fill=blue!30] (M1) circle (\r) node[right=0.3cm] {$m_1$};
        \filldraw[fill=green!30] (M2) circle (\r) node[right=0.3cm] {$m_2$};

        % Angles
        % Angle 1

    \end{tikzpicture}

    \label{fig:double_pendulum}
\end{figure}

The vector from the origin to the upper point mass is defined as $\mathbf{r}_1$ and the vector from the origin to the lower point mass is defined as $\mathbf{r}_2$. In terms of $\theta_1$ and $\theta_2$, these vectors and their first and second derivatives are:

\begin{align*}
    \mathbf{r}_1 &= l_1 \sin{\theta_1} \hat{\imath} - l_1 \cos{\theta_1} \hat{\jmath} \\
    \mathbf{r}_2 &= \left( l_1 \sin{\theta_1} + l_2 \sin{\theta_2} \right) \hat{\imath} - \left( l_1 \cos{\theta_1} + l_2 \cos{\theta_2} \right) \hat{\jmath} \\[3mm]
    \dot{\mathbf{r}}_1 &= \dot{\theta}_1 l_1 \cos{\theta_1} \hat{\imath} + \dot{\theta}_1 l_1 \sin{\theta_1} \hat{\jmath} \\
    \dot{\mathbf{r}}_2 &= \left( \dot{\theta}_1 l_1 \cos{\theta_1} + \dot{\theta}_2 l_2 \cos{\theta_2} \right) \hat{\imath} + \left( \dot{\theta}_1 l_1 \sin{\theta_1} + \dot{\theta}_2 l_2 \sin{\theta_2} \right) \hat{\jmath} \\[3mm]
    \ddot{\mathbf{r}}_1 &= \left( \ddot{\theta}_1 l_1 \cos{\theta_1} - \dot{\theta}_1^2 l_1 \sin{\theta_1} \right) \hat{\imath} + \left( \ddot{\theta}_1 l_1 \sin{\theta_1} + \dot{\theta}_1^2 l_1 \cos{\theta_1} \right) \hat{\jmath} \\
    \ddot{\mathbf{r}}_2 &= \left( \ddot{\theta}_1 l_1 \cos{\theta_1} - \dot{\theta}_1^2 l_1 \sin{\theta_1} + \ddot{\theta}_2 l_2 \cos{\theta_2} - \dot{\theta}_2^2 l_2 \sin{\theta_2} \right) \hat{\imath} \\ 
    &+ \left( \ddot{\theta}_1 l_1 \sin{\theta_1} + \dot{\theta}_1^2 l_1 \cos{\theta_1} + \ddot{\theta}_2 l_2 \sin{\theta_2} + \dot{\theta}_2^2 l_2 \cos{\theta_2} \right) \hat{\jmath}
\end{align*}

The double pendulum is then split into two components. Figure~\ref{fig:double pendulum fbds} shows the two free body diagrams, including the interaction forces between the two components and the inertial forces due to acceleration.

The sum of moments about the pivot for Component 1, the pivot of Component 2, and the point mass of Component 2 respectively give us the following equations, noting that $F_{12} = - F_{21}$ and dividing by the lengths of the rods to simplify:

\begin{align*}
    \sum M^*_{O_1} &= 0 = \cos(\theta_1) F_{21,x} + \ sin(\theta_1) F_{21,y} - m_1 \cos(\theta_1) \ddot{r}_{1,x} - m_1 \sin(\theta_1) \ddot{r}_{1,y} - \sin(\theta_1) m_1 g \\
    \sum M^*_{O_2} &= 0 = - m_2 \cos(\theta_2) \ddot{r}_{2,x} - m_2 \sin(\theta_2) \ddot{r}_{2,y} - \sin(\theta_2) m_2 g \\
    \sum M^*_{m_2} &= 0 = - \cos(\theta_2) F_{21,x} - \ sin(\theta_2) F_{21,y}
\end{align*}
